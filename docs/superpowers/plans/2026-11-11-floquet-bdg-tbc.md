```python
def step_evolution(k0, k1, period, time):
    if period <= 0 or not 0 <= time <= period:
        raise ValueError("require period > 0 and 0 <= time <= period")
    if time == 0:
        return np.eye(k0.shape[0], dtype=complex)
    return expm(-4*k1) @ expm(-4*k0*time)

def floquet_gap(u, epsilon):
    if epsilon not in (0.0, np.pi):
        raise ValueError("epsilon must be 0 or pi")
    return float(np.min(np.abs(np.linalg.eigvals(u) - np.exp(1j*epsilon))))
```

The complete `twisted_majorana_k` starts from `majorana_k(n, mean(mu))`, writes onsite `-mu_j/2` pairs, then adds only the four phased closing-bond entries from the design. Internal bonds remain the Fock-checked entries. Require `n>=2` and a scalar or length-`n` chemical potential.

```python
values, vectors = np.linalg.eig(u_step_T)
phi = -np.angle(values)
gap = float(np.min(np.abs(values - np.exp(1j*epsilon))))
if gap <= gap_tolerance:
    return {"status": "gapless_or_undefined", "bulk_gap": gap,
            "value": None, "reason": "bulk_gap_closed"}
phi_eff = np.where(phi < 0, 2*epsilon - np.abs(phi), np.abs(phi) - 2*np.pi)
factors = np.exp(1j*phi_eff*time/period)
# vectors come from a unitary Schur decomposition, not generic eig.
power = (vectors*factors) @ vectors.conj().T
u_epsilon = u_step_t @ power
return {"status": "ok", "u": u_epsilon, "bulk_gap": gap,
        "phi": phi, "phi_eff": phi_eff}
```
